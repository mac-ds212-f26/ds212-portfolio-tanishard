# Load built-in dataset
data(mtcars)

# Preserve car names
mtcars$car <- rownames(mtcars)

# Create interactive scatterplot
fig <- plot_ly(
  data = mtcars,
  x = ~hp,
  y = ~mpg,
  color = ~factor(cyl),
  size = ~wt,
  text = ~car,
  type = "scatter",
  mode = "markers",
  hovertemplate = paste(
    "<b>%{text}</b><br>",
    "Horsepower: %{x}<br>",
    "MPG: %{y}<br>",
    "<extra></extra>"
  )
)

fig <- fig %>%
  layout(
    title = "Horsepower vs Fuel Efficiency",
    xaxis = list(title = "Horsepower"),
    yaxis = list(title = "Miles per Gallon"),
    legend = list(title = list(text = "Cylinders"))
  )

# Load built-in dataset
data(iris)

# Create interactive scatterplot
fig <- plot_ly(
  data = iris,
  x = ~Petal.Length,
  y = ~Petal.Width,
  color = ~Species,
  type = "scatter",
  mode = "markers",
  marker = list(size = 10),
  hovertemplate = paste(
    "Species: %{fullData.name}<br>",
    "Petal Length: %{x}<br>",
    "Petal Width: %{y}<br>",
    "<extra></extra>"
  )
)

# Customize layout
fig <- fig %>%
  layout(
    title = "Iris Flowers: Petal Length vs. Petal Width",
    xaxis = list(title = "Petal Length (cm)"),
    yaxis = list(title = "Petal Width (cm)"),
    legend = list(title = list(text = "Species"))
  )

# Create the initial plot
fig <- plot_ly(
  data = iris,
  x = ~Petal.Length,
  y = ~Petal.Width,
  color = ~Species,
  colors = c(
    "setosa" = "#F8766D",
    "versicolor" = "#00BA38",
    "virginica" = "#619CFF"
  ),
  type = "scatter",
  mode = "markers",
  marker = list(size = 9),
  hovertemplate = paste(
    "Species: %{fullData.name}<br>",
    "Petal Length: %{x}<br>",
    "Measurement: %{y}<br>",
    "<extra></extra>"
  )
)

# Add dropdown menu
fig <- fig %>%
  layout(
    title = "Explore Iris Flower Measurements",

    xaxis = list(
      title = "Petal Length (cm)"
    ),

    yaxis = list(
      title = "Petal Width (cm)"
    ),

    updatemenus = list(
      list(
        type = "dropdown",
        x = 1.05,
        y = 1,
        buttons = list(

          list(
            method = "update",
            args = list(
              list(y = list(iris$Petal.Width)),
              list(yaxis = list(title = "Petal Width (cm)"))
            ),
            label = "Petal Width"
          ),

          list(
            method = "update",
            args = list(
              list(y = list(iris$Sepal.Width)),
              list(yaxis = list(title = "Sepal Width (cm)"))
            ),
            label = "Sepal Width"
          ),

          list(
            method = "update",
            args = list(
              list(y = list(iris$Sepal.Length)),
              list(yaxis = list(title = "Sepal Length (cm)"))
            ),
            label = "Sepal Length"
          )
        )
      )
    )
  )
